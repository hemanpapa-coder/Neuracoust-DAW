--- /usr/local/bin/neuracoust-rt-audio-server
#!/usr/bin/env bash
set -euo pipefail

STATE_DIR="${STATE_DIR:-/run/neuracoust-rt-audio}"
LOG_DIR="${LOG_DIR:-/var/log/neuracoust-rt-audio}"
CONFIG_FILE="${CONFIG_FILE:-/etc/neuracoust-rt-audio/server.conf}"

mkdir -p "$STATE_DIR" "$LOG_DIR"
[[ -f "$CONFIG_FILE" ]] && source "$CONFIG_FILE"

SAMPLE_RATE="${SAMPLE_RATE:-48000}"
PERIOD_SIZE="${PERIOD_SIZE:-128}"
NET_BUFFER="${NET_BUFFER:-3}"
NETJACK_PORT="${NETJACK_PORT:-19000}"
NETJACK_MCAST="${NETJACK_MCAST:-225.3.19.154}"
CHANNELS="${CHANNELS:-64}"
DSP_CPU_MASK="${DSP_CPU_MASK:-2,3}"
PRIORITY="${PRIORITY:-95}"

stop_existing() {
  pkill -x jackd 2>/dev/null || true
  sleep 0.5
}

case "${1:-start}" in
  start)
    stop_existing
    echo "Starting Neuracoust RT Audio Server: ${SAMPLE_RATE}Hz/${PERIOD_SIZE}, channels=${CHANNELS}, cpu=${DSP_CPU_MASK}"
    taskset -c "$DSP_CPU_MASK" jackd \
      -R -P "$PRIORITY" \
      -d dummy \
      -r "$SAMPLE_RATE" \
      -p "$PERIOD_SIZE" \
      -C "$CHANNELS" \
      -P "$CHANNELS" \
      -m &
    JACKD_PID=$!
    echo "$JACKD_PID" > "$STATE_DIR/jackd.pid"
    jack_wait -w -t 5 >/dev/null 2>&1 || true
    export JACK_NETJACK_MULTICAST="$NETJACK_MCAST"
    export JACK_NETJACK_PORT="$NETJACK_PORT"
    jack_load netmanager >/dev/null 2>&1 || true
    wait "$JACKD_PID"
    ;;
  status)
    pgrep -a jackd || exit 1
    jack_bufsize 2>/dev/null || true
    ;;
  stop)
    stop_existing
    ;;
  *)
    echo "Usage: $0 {start|stop|status}" >&2
    exit 2
    ;;
esac

#!/usr/bin/env bash
set -euo pipefail

CONFIG_FILE="${CONFIG_FILE:-/etc/neuracoust-rt-dsp/server.conf}"
[[ -f "$CONFIG_FILE" ]] && source "$CONFIG_FILE"

INSTALL_DIR="${INSTALL_DIR:-/opt/neuracoust/rt_engine}"
MODULE_PATH="${MODULE_PATH:-${INSTALL_DIR}/modules/na_4001e.so}"
NA_RT_AUDIO_PORT="${NA_RT_AUDIO_PORT:-20000}"
NA_RT_MONITOR_PORT="${NA_RT_MONITOR_PORT:-20001}"
DSP_CPU_MASK="${DSP_CPU_MASK:-2,3}"

case "${1:-start}" in
  start)
    exec taskset -c "$DSP_CPU_MASK" "$INSTALL_DIR/neuracoust-rt-engine" \
      --module "$MODULE_PATH" \
      --port "$NA_RT_AUDIO_PORT" \
      --monitor-port "$NA_RT_MONITOR_PORT"
    ;;
  self-test)
    exec "$INSTALL_DIR/neuracoust-rt-engine" --module "$MODULE_PATH" --self-test
    ;;
  *)
    echo "Usage: $0 {start|self-test}" >&2
    exit 2
    ;;
esac

#!/usr/bin/env bash
set -euo pipefail

echo "=== Neuracoust RT Audio Check ==="
echo

echo "[Identity]"
hostnamectl 2>/dev/null || hostname
echo

echo "[Kernel]"
uname -a
echo "cmdline: $(cat /proc/cmdline)"
if uname -a | grep -Eqi 'preempt_rt|rt-amd64| PREEMPT_RT'; then
  echo "rt-kernel: yes"
else
  echo "rt-kernel: not detected"
fi
echo

echo "[CPU]"
lscpu | sed -n '1,32p'
echo "isolated: $(cat /sys/devices/system/cpu/isolated 2>/dev/null || true)"
for governor in /sys/devices/system/cpu/cpu[0-9]*/cpufreq/scaling_governor; do
  [[ -f "$governor" ]] && echo "$governor=$(cat "$governor")"
done
echo

echo "[Realtime Limits]"
ulimit -r || true
ulimit -l || true
echo

echo "[Audio Packages]"
dpkg -l linux-image-rt-amd64 jackd2 2>/dev/null | sed -n '1,80p' || true
echo

echo "[Service]"
systemctl --no-pager --full status neuracoust-rt-audio.service 2>/dev/null | sed -n '1,80p' || true
echo

echo "[JACK]"
pgrep -a jackd || true
sudo jack_bufsize 2>/dev/null || jack_bufsize 2>/dev/null || true
sudo jack_lsp 2>/dev/null | sed -n '1,40p' || jack_lsp 2>/dev/null | sed -n '1,40p' || true
echo

echo "[Network]"
ip -br addr
ss -lunpt 2>/dev/null | grep -E '19000|20000|20001|20009|20010' || true
echo

echo "[Recent Logs]"
journalctl -u neuracoust-rt-audio.service -n 80 --no-pager 2>/dev/null || true

--- /usr/local/sbin/neuracoust-console-status
#!/usr/bin/env bash
trap "" INT TERM TSTP

while true; do
  clear

  host=$(hostname)
  ip=$(hostname -I 2>/dev/null | awk '{print $1}')
  dsp=$(systemctl is-active neuracoust-rt-dsp.service 2>/dev/null || true)
  if ss -lunpt 2>/dev/null | grep -q ':20000 '; then
    port=open
  else
    port=closed
  fi
  now=$(date '+%Y-%m-%d %H:%M:%S %Z')
  up=$(uptime -p 2>/dev/null | sed 's/^up //')

  printf '\n'
  printf '  누라쿠스트 DSP 서버\n'
  printf '  ========================\n\n'

  if [ "$dsp" = active ] && [ "$port" = open ]; then
    printf '  STATUS : STANDBY / READY\n'
  else
    printf '  STATUS : CHECK SERVICE\n'
  fi

  printf '  HOST   : %s\n' "$host"
  printf '  IP     : %s\n' "${ip:-not-ready}"
  printf '  DSP    : %s\n' "${dsp:-unknown}"
  printf '  UDP    : 20000/%s\n' "$port"
  printf '  UPTIME : %s\n' "$up"
  printf '  TIME   : %s\n\n' "$now"
  printf '  Remote control: SSH / 누라쿠스트 DAW\n'
  printf '  Local login is disabled on this display.\n\n'

  sleep 5
done
